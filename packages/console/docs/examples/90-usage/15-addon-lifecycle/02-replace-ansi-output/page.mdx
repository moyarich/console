---
label: "Replace ANSI output"
description: "Usage example for replace ansi output."
---

See the [API reference](#/package-console/reference/addons/overview) for the contract and options.

```tsx
function createTerminalSurfaceAddon(): ConsoleAddon {
  return {
    id: "@acme/console-terminal-surface",
    activate(host) {
      host.extensions.register(consoleExtensionPoints.outputRenderer, {
        mode: "ansi",
        render(context) {
          if (context.mode !== "ansi") {
            return undefined;
          }

          return <MyTerminal entries={context.entries} />;
        },
      });
    },
  };
}
```
