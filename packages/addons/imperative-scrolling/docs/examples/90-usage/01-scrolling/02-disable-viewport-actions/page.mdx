---
label: "Disable viewport actions"
description: "Usage example for disable viewport actions."
---

See the [API reference](#/addons/imperative-scrolling/reference/scrolling) for the contract and options.

```tsx
import {
  createImperativeScrollingAddon,
  IMPERATIVE_SCROLLING_ADDON_ID,
} from "@moyarich/console-addon-imperative-scrolling";

const scrollingAddon = createImperativeScrollingAddon();

<Console
  messages={messages}
  addons={[scrollingAddon]}
  disabledAddonIds={[IMPERATIVE_SCROLLING_ADDON_ID]}
/>;
```
